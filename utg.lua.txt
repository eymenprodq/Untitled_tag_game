local function MakeColor(R, G, B)
    return Color3.fromRGB(R, G, B)
end

local function UpdateHitboxes()
    for _, Other in next, Players:GetPlayers() do
        if Other ~= Player and Other.Character then
            local HRP = Other.Character:FindFirstChild("HumanoidRootPart")
            if HRP then
                if not OriginalSizes[Other] then
                    OriginalSizes[Other] = HRP.Size
                end
                HRP.Size = Vector3.new(2 * HitboxScale, 2 * HitboxScale, 1 * HitboxScale)
                HRP.Transparency = 0.7
                HRP.CanCollide = false
            end
        end
    end
end

local function ResetHitboxes()
    for _, Other in next, Players:GetPlayers() do
        if Other ~= Player and Other.Character then
            local HRP = Other.Character:FindFirstChild("HumanoidRootPart")
            if HRP then
                if OriginalSizes[Other] then
                    HRP.Size = OriginalSizes[Other]
                    OriginalSizes[Other] = nil
                else
                    HRP.Size = Vector3.new(2, 2, 1)
                end
                HRP.Transparency = 0
                HRP.CanCollide = true
            end
        end
    end
    OriginalSizes = {}
end

local function ClickTagButton()
    local PG = Player:FindFirstChild("PlayerGui")
    if not PG then return false end
    local Touch = PG:FindFirstChild("TouchGuiUTG")
    if not Touch then return false end
    local UTG = Touch:FindFirstChild("UTGButtons")
    if not UTG then return false end
    local Btn = UTG:FindFirstChild("TagButton")
    if not Btn then return false end
    local Connections = getconnections(Btn.Activated)
    if not Connections or #Connections == 0 then return false end
    for Index = 1, #Connections do
        local Connection = Connections[Index]
        if Connection then
            Connection:Fire()
            return true
        end
    end
    return false
end

local function GetRemote(Name)
    for _, V in next, getgc(true) do
        if typeof(V) == "Instance" and V:IsA("RemoteEvent") and V.Name == Name then
            return V
        end
        if typeof(V) == "Instance" and V:IsA("RemoteFunction") and V.Name == Name then
            return V
        end
    end
    return nil
end

local function TagAuraTick()
    if tick() - LastTagTime < AuraCooldown then return end
    local MyChar = Player.Character
    if not MyChar then return end
    local MyHRP = MyChar:FindFirstChild("HumanoidRootPart")
    if not MyHRP then return end

    local ClosestTarget = nil
    local ClosestDistance = AuraRange
    for _, Other in next, Players:GetPlayers() do
        if Other ~= Player and Other.Character then
            local HRP = Other.Character:FindFirstChild("HumanoidRootPart")
            if HRP then
                local Dist = (MyHRP.Position - HRP.Position).Magnitude
                if Dist <= ClosestDistance then
                    ClosestDistance = Dist
                    ClosestTarget = Other
                end
            end
        end
    end

    if not ClosestTarget then return end

    if ClickTagButton() then
        LastTagTime = tick()
        return
    end

    local Force = GetRemote("ForcePlayerTag")
    if Force then
        Force:FireServer(ClosestTarget)
        LastTagTime = tick()
    end
end

local function DestroyEsp(Obj)
    if not Obj or not Obj.Connected then return end
    pcall(function() Obj.Box:Remove() end)
    pcall(function() Obj.Outline:Remove() end)
    pcall(function() Obj.Fill:Remove() end)
    pcall(function() Obj.Name:Remove() end)
    pcall(function() Obj.HealthBG:Remove() end)
    pcall(function() Obj.HealthBar:Remove() end)
    pcall(function() Obj.HealthText:Remove() end)
    pcall(function() Obj.DistanceText:Remove() end)
    pcall(function() Obj.Tracer:Remove() end)
    Obj.Connected = false
end

local function HideEsp(Obj)
    if not Obj or not Obj.Connected then return end
    Obj.Box.Visible = false
    Obj.Outline.Visible = false
    Obj.Fill.Visible = false
    Obj.Name.Visible = false
    Obj.HealthBG.Visible = false
    Obj.HealthBar.Visible = false
    Obj.HealthText.Visible = false
    Obj.DistanceText.Visible = false
    Obj.Tracer.Visible = false
end

local function UpdateEsp()
    if DrawingAvailable == nil then
        DrawingAvailable = pcall(function()
            local Test = Drawing.new("Square")
            Test:Remove()
            return true
        end)
        if not DrawingAvailable then
            EspEnabled = false
            return
        end
    end
    if not DrawingAvailable then return end

    local ActivePlayers = {}

    for _, Other in next, Players:GetPlayers() do
        if Other ~= Player then
            local Char = Other.Character
            local HasValidChar = Char
                and Char:FindFirstChild("HumanoidRootPart")
                and Char:FindFirstChild("Humanoid")
                and Char.Humanoid.Health > 0

            if HasValidChar then
                ActivePlayers[Other] = true

                if not EspCache[Other] or not EspCache[Other].Connected then
                    EspCache[Other] = CreateEspObject()
                end
                local Obj = EspCache[Other]

                if Obj.Connected then
                    local HRP = Char.HumanoidRootPart
                    local Hum = Char.Humanoid
                    local MyChar = Player.Character
                    local MyHRP = MyChar and MyChar:FindFirstChild("HumanoidRootPart")

                    local Pos, OnScreen = workspace.CurrentCamera:WorldToViewportPoint(HRP.Position)
                    if not OnScreen then
                        HideEsp(Obj)
                    else
                        local Dist = MyHRP and (MyHRP.Position - HRP.Position).Magnitude or 0
                        if Dist > EspMaxDistance then
                            HideEsp(Obj)
                        else
                            local FOV = workspace.CurrentCamera.FieldOfView
                            if FOV <= 0 then FOV = 70 end
                            local ScaleFactor = 1 / (Pos.Z * math.tan(math.rad(FOV * 0.5)) * 2)
                            local BoxSize = Vector2.new(4, 6) * ScaleFactor * 600
                            local Width = math.clamp(BoxSize.X, 20, 400)
                            local Height = math.clamp(BoxSize.Y, 30, 600)
                            local X = Pos.X - Width / 2
                            local Y = Pos.Y - Height / 2

                            Obj.Box.Size = Vector2.new(Width, Height)
                            Obj.Box.Position = Vector2.new(X, Y)
                            Obj.Box.Visible = EspBox

                            Obj.Outline.Size = Vector2.new(Width, Height)
                            Obj.Outline.Position = Vector2.new(X, Y)
                            Obj.Outline.Visible = EspBox and EspOutline

                            Obj.Fill.Size = Vector2.new(Width, Height)
                            Obj.Fill.Position = Vector2.new(X, Y)
                            Obj.Fill.Visible = EspFilled

                            Obj.Name.Position = Vector2.new(Pos.X, Y - 14)
                            Obj.Name.Text = Other.DisplayName
                            Obj.Name.Visible = EspNames

                            local HP_Pct = math.clamp(Hum.Health / Hum.MaxHealth, 0, 1)
                            local BarWidth = 2
                            local BarX = X - BarWidth - 3
                            Obj.HealthBG.Size = Vector2.new(BarWidth, Height)
                            Obj.HealthBG.Position = Vector2.new(BarX, Y)
                            Obj.HealthBG.Visible = EspHealth

                            Obj.HealthBar.Size = Vector2.new(BarWidth, Height * HP_Pct)
                            Obj.HealthBar.Position = Vector2.new(BarX, Y + Height * (1 - HP_Pct))
                            Obj.HealthBar.Color = Color3.fromHSV(HP_Pct * 0.33, 1, 1)
                            Obj.HealthBar.Visible = EspHealth

                            Obj.HealthText.Position = Vector2.new(BarX - 32, Y + Height - 14)
                            Obj.HealthText.Text = tostring(math.floor(Hum.Health))
                            Obj.HealthText.Visible = EspHealth

                            Obj.DistanceText.Position = Vector2.new(Pos.X, Y + Height + 2)
                            Obj.DistanceText.Text = tostring(math.floor(Dist)) .. " studs"
                            Obj.DistanceText.Visible = EspDistance

                            Obj.Tracer.From = Vector2.new(
                                workspace.CurrentCamera.ViewportSize.X / 2,
                                workspace.CurrentCamera.ViewportSize.Y
                            )
                            Obj.Tracer.To = Vector2.new(Pos.X, Pos.Y + Height / 2)
                            Obj.Tracer.Visible = EspTracers
                        end
                    end
                end
            else
                if EspCache[Other] and EspCache[Other].Connected then
                    HideEsp(EspCache[Other])
                end
            end
        end
    end

    for CachedPlayer, Obj in pairs(EspCache) do
        if not ActivePlayers[CachedPlayer] then
            DestroyEsp(Obj)
            EspCache[CachedPlayer] = nil
        end
    end
end

local function HideAllEsp()
    for _, Obj in pairs(EspCache) do
        HideEsp(Obj)
    end
end

local function RefreshEspVisibility()
    if not EspEnabled then return end
    UpdateEsp()
end

local function GetCharacter()
    return Player.Character
end

local function GetHumanoid()
    local Char = GetCharacter()
    return Char and Char:FindFirstChildOfClass("Humanoid")
end

local function GetRootPart()
    local Char = GetCharacter()
    return Char and Char:FindFirstChild("HumanoidRootPart")
end

local function DoFling()
    local HRP = GetRootPart()
    if not HRP then return end
    HRP.AssemblyLinearVelocity = UltimateTechFlingVector
end

UserInputService.JumpRequest:Connect(function()
    if InfJumpEnabled then
        local Humanoid = GetHumanoid()
        if Humanoid then
            Humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end
    end
end)

UserInputService.InputBegan:Connect(function(Input, Processed)
    if Processed then return end
    if UltimateTechEnabled and Input.KeyCode == UltimateTechKey then
        DoFling()
    end
    if FloatEnabled then
        if Input.KeyCode == Enum.KeyCode.E then
            FloatUp = true
        elseif Input.KeyCode == Enum.KeyCode.R then
            FloatDown = true
        end
    end
end)

UserInputService.InputEnded:Connect(function(Input, Processed)
    if Processed then return end
    if FloatEnabled then
        if Input.KeyCode == Enum.KeyCode.E then
            FloatUp = false
        elseif Input.KeyCode == Enum.KeyCode.R then
            FloatDown = false
        end
    end
end)

RunService.RenderStepped:Connect(function(Delta)
    pcall(function()
        if HitboxEnabled and HitboxScale > 1 then
            UpdateHitboxes()
        elseif not HitboxEnabled then
            ResetHitboxes()
        end

        if AuraEnabled then
            TagAuraTick()
        end

        if EspEnabled then
            UpdateEsp()
        end
        
        if CFrameWalkEnabled then
            local Hum = GetHumanoid()
            local HRP = GetRootPart()
            if Hum and HRP and Hum.MoveDirection.Magnitude > 0 then
                HRP.CFrame = HRP.CFrame + (Hum.MoveDirection * (CFrameWalkSpeed * Delta))
            end
        end

        if FloatEnabled then
            local HRP = GetRootPart()
            if HRP then
                if not FloatBV or FloatBV.Parent ~= HRP then
                    if FloatBV then FloatBV:Destroy() end
                    FloatBV = Instance.new("BodyVelocity")
                    FloatBV.MaxForce = Vector3.new(0, 100000, 0)
                    FloatBV.Velocity = Vector3.new(0, 0, 0)
                    FloatBV.Parent = HRP
                end
                local VelY = 0
                if FloatUp then VelY = FloatSpeed end
                if FloatDown then VelY = -FloatSpeed end
                FloatBV.Velocity = Vector3.new(0, VelY, 0)
            end
        else
            if FloatBV then
                FloatBV:Destroy()
                FloatBV = nil
            end
        end
    end)
end)
